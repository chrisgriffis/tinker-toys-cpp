#include <iostream>
#include <utility>
#include <vector>
#include <queue>

using namespace std;

int main()
{
    auto idx_from_ints = [](int r, int c) {return 8 * r + c; };
    auto idx_from_pair = [](pair<int, int> p) {return 8 * p.first + p.second; };
    auto pair_from_idx = [](int i) {return make_pair(i / 8, i % 8); };

    vector<vector<int>> adj(64, vector<int>{});

    pair<int, int> offsets[8] = {
        make_pair(1,2),
        make_pair(-1,2),
        make_pair(1,-2),
        make_pair(-1,-2),

        make_pair(2,1),
        make_pair(-2,1),
        make_pair(2,-1),
        make_pair(-2,-1)
    };

    for (int idx = 0; idx<64; ++idx)
    {
        for (auto o : offsets)
        {
            auto curr = pair_from_idx(idx);
            int r = curr.first + o.first;
            int c = curr.second + o.second;
            if ((r >= 0) && (r < 8) && (c >= 0) && (c < 8))
                adj[idx].emplace_back(idx_from_ints(r,c));
        }
    }

    auto a = make_pair(1, 1);
    auto b = make_pair(2, 2);

    auto steps = [idx_from_pair,adj](auto a,auto b) {
        queue<int> q1, q2;
        queue<int>* currq(&q1);
        queue<int>* otherq(&q2);
        currq->push(idx_from_pair(a));
        int count = 0;

        while (!currq->empty() || !otherq->empty())
        {
            while (!currq->empty())
            {
                if (currq->front() == idx_from_pair(b))
                    return count;
                for (int e : adj[currq->front()])
                    otherq->emplace(e);
                currq->pop();
            }
            ++count;
            swap(currq, otherq);
        }
        return -1;
    };

    cout << "steps: " << steps(a, b);

	int sdfgwert;
	cin >> sdfgwert;
    return 0;
}
